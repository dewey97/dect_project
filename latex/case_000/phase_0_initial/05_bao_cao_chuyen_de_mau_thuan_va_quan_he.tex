\section*{KẾT QUẢ RÀ SOÁT QUAN HỆ VÀ MÂU THUẪN SƠ BỘ}

\begin{enumerate}[label=\arabic*., leftmargin=1.5cm]
    \item \textbf{Nhóm 1: Trần Ngọc Mai (Em họ nạn nhân)}
    \begin{itemize}[leftmargin=0.8cm]
        \item \textit{Mối quan hệ:} Thuộc hàng thừa kế thứ hai. Sau khi ông nội mất cách đây 03 tháng, giữa Mai và Khang có phát sinh bất đồng về quyền quản lý, sử dụng căn nhà số 14 Đường Bờ Sông (nằm trong diện dự án quy hoạch đền bù).
        \item \textit{Vấn đề cần làm rõ:} Mục đích chuyến thăm nhà Khang chiều ngày 24/07/2026 và các tranh chấp tài sản liên quan.
    \end{itemize}

    \item \textbf{Nhóm 2: Lê Quang Vũ (Chồng của Mai)}
    \begin{itemize}[leftmargin=0.8cm]
        \item \textit{Mối quan hệ:} Cán bộ đo đạc địa chính. Vũ từng tham gia đo vẽ hiện trạng khu vực đền bù Phân khu Cảng và có giao dịch tài chính cá nhân chưa rõ bản chất với nạn nhân Khang.
        \item \textit{Vấn đề cần làm rõ:} Lịch trình di chuyển trong tối 24/07 và các cam kết tài chính giữa Vũ và nạn nhân.
    \end{itemize}

    \item \textbf{Nhóm 3: Nguyễn Thanh Tùng (Bạn từ thời thơ ấu)}
    \begin{itemize}[leftmargin=0.8cm]
        \item \textit{Mối quan hệ:} Bạn học cũ cùng khu phố. Tùng mới trở về địa phương gần đây sau thời gian dài làm ăn xa. Hai gia đình trong quá khứ từng có biến cố cá nhân.
        \item \textit{Vấn đề cần làm rõ:} Lý do Tùng xuất hiện tại khu vực ngõ 14 Đường Bờ Sông trong khung giờ xảy ra sự việc.
    \end{itemize}

    \item \textbf{Nhóm 4: Trần Thị Hà (Bạn gái cũ)}
    \begin{itemize}[leftmargin=0.8cm]
        \item \textit{Mối quan hệ:} Có quan hệ tình cảm gắn bó lâu năm với Khang, gần đây có thông tin hai người xảy ra bất đồng rạn nứt.
        \item \textit{Vấn đề cần làm rõ:} Xác minh chứng cứ ngoại phạm và các xung đột tình cảm phát sinh trước thời điểm án mạng.
    \end{itemize}
\end{enumerate}
