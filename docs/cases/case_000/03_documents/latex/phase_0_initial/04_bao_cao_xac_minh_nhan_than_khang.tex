\section*{I. THÔNG TIN LÝ LỊCH VÀ HOÀN CẢNH GIA ĐÌNH}
\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Họ và tên:} NGUYỄN VĂN KHANG. Tên gọi khác: Khang "Đen".
    \item \textbf{Ngày sinh:} 15/06/1990. Nguyên quán: Nam Định.
    \item \textbf{CCCD số:} 001090003412 cấp ngày 10/05/2021 tại Cục CSQLHC về TTXH.
    \item \textbf{Nơi ĐKHKTT \& Thường trú:} Số 14 Đường Bờ Sông, Phân khu Cảng, TP. Hà Nội.
    \item \textbf{Hoàn cảnh gia đình:} Bố mẹ mất sớm năm 2005. Khang sống từ nhỏ với ông nội (cụ Nguyễn Văn Thành). Cụ Thành qua đời tháng 04/2026. Hiện Khang sống một mình tại căn nhà 75m2 do ông nội để lại.
\end{itemize}

\section*{II. QUÁ TRÌNH HỌC TẬP, VIỆC LÀM VÀ HOẠT ĐỘNG TÀI CHÍNH}
\begin{enumerate}[label=\arabic*., leftmargin=1.5cm]
    \item \textbf{Học vấn \& Quá trình công tác:} Tốt nghiệp Trung cấp Xây dựng Hà Nội năm 2012. Không công tác tại cơ quan, doanh nghiệp cố định nào. 
    \item \textbf{Hoạt động thực tế:} Từ năm 2018 đến nay làm dịch vụ "Tư vấn tài chính nhanh" (thực chất là cho vay bốc họ / vay nặng lãi lãi suất cao). Khang đứng ra thu tiền họ hàng ngày, chuyên cầm cố giấy tờ nhà đất khu vực ven sông.
    \item \textbf{Tình trạng cư trú \& Quản lý tài sản:} Đăng ký hộ khẩu thường trú liên tục tại số 14 Bờ Sông. Hiện đứng tên đại diện quản lý nhà đất đền bù đợt 1 dự án quy hoạch giải tỏa Bờ Sông.
    \item \textbf{Tiền án tiền sự \& Mối quan hệ phực tạp:} Chưa có tiền án tiền sự. Tuy nhiên từng bị phản ánh nhiều lần về hành vi đe dọa đòi nợ thuê, có mối quan hệ mâu thuẫn phức tạp với nhiều người vay tiền bốc họ tại địa phương.
\end{enumerate}
