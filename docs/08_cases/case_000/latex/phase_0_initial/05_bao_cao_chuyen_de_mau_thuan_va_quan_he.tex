\section*{MA TRẬN MÂU THUẪN VÀ ĐỘNG CƠ NGHI PHẠM}

\begin{enumerate}[label=\arabic*., leftmargin=1.5cm]
    \item \textbf{Nhóm 1: Trần Ngọc Mai (Em họ nạn nhân)}
    \begin{itemize}[leftmargin=0.8cm]
        \item \textit{Mâu thuẫn chính:} Tranh chấp di chúc đền bù căn nhà số 14 Bờ Sông.
        \item \textit{Động cơ:} Nghi ngờ Khang làm giả di chúc cướp toàn bộ tiền đền bù 3.5 tỷ đồng của dòng họ.
    \end{itemize}

    \item \textbf{Nhóm 2: Lê Quang Vũ (Chồng của Mai)}
    \begin{itemize}[leftmargin=0.8cm]
        \item \textit{Mâu thuẫn chính:} Gian lận đo đạc đền bù & Khoản nợ bí mật 350 triệu.
        \item \textit{Động cơ:} Bị Khang tống tiền ép đo khống diện tích đất, muốn tiêu hủy chứng cứ bản vẽ gian lận.
    \end{itemize}

    \item \textbf{Nhóm 3: Nguyễn Thanh Tùng (Bạn từ thời thơ ấu)}
    \begin{itemize}[leftmargin=0.8cm]
        \item \textit{Mâu thuẫn chính:} Bi kịch ngạt khí em trai (Gia Huy) năm 1998 & Hành vi tống tiền.
        \item \textit{Động cơ:} Thù hằn thâm sầu 12 năm vì Khang gài chốt tủ gây ra cái chết của em trai và dùng kỷ vật đe dọa.
    \end{itemize}

    \item \textbf{Nhóm 4: Trần Thị Hà (Bạn gái cũ)}
    \begin{itemize}[leftmargin=0.8cm]
        \item \textit{Mâu thuẫn chính:} Tình cảm ghen tuông bộc phát.
        \item \textit{Động cơ:} Yêu mù quáng 3 năm nhưng bị Khang phớt lờ ruồng bỏ để chạy theo tình nhân mới.
    \end{itemize}
\end{enumerate}
