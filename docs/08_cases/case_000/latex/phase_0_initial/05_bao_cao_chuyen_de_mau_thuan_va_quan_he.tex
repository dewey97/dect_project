\section*{MA TRẬN 04 NGHI PHẠM CHÍNH}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!black,title=\textbf{1. Trần Ngọc Mai (36 tuổi) - Em họ nạn nhân}]
\begin{itemize}
    \item \textbf{Mâu thuẫn:} Tranh chấp di chúc thừa kế căn nhà 14 Bờ Sông.
    \item \textbf{Động cơ:} Giết Khang để đòi lại quyền thừa kế tài sản đền bù.
    \item \textbf{Nghi vấn:} Dấu vân tay có trên hộp sắt tủ âm tường.
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=blue!5!white,colframe=blue!75!black,title=\textbf{2. Lê Quang Vũ (39 tuổi) - Chồng Mai / Cán bộ đo đạc}]
\begin{itemize}
    \item \textbf{Mâu thuẫn:} Nợ Khang 350 triệu giấu Mai, bị Khang ép đo khống 45m2 đất rồi tống tiền.
    \item \textbf{Động cơ:} Tiêu hủy bản vẽ chứng cứ và xóa nợ cá nhân.
    \item \textbf{Nghi vấn:} Xóa lịch sử cuộc gọi, dấu giày nam tại cửa sau.
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=yellow!10!white,colframe=yellow!75!black,title=\textbf{3. Tùng (35 tuổi) - Bạn cũ / Anh trai Gia Huy}]
\begin{itemize}
    \item \textbf{Mâu thuẫn:} Bị Khang tống tiền đe dọa bí mật tai nạn trốn tìm năm xưa.
    \item \textbf{Động cơ:} Trả thù cho em trai và thủ tiêu kẻ tống tiền.
    \item \textbf{Nghi vấn:} Camera cây xăng ghi nhận xuất hiện từ 19:45 đến 20:15.
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=red!5!white,colframe=red!75!black,title=\textbf{4. Trần Thị Hà (35 tuổi) - Bạn gái cũ nạn nhân}]
\begin{itemize}
    \item \textbf{Mâu thuẫn:} Yêu cuồng loạn, vừa bị Khang phớt lờ đòi chia tay để quen nhân tình mới.
    \item \textbf{Động cơ:} Ghen tuông bộc phát, uất ức bị bỏ rơi.
    \item \textbf{Nghi vấn:} Âm thầm bám theo Khang, dấu giày nữ gần cửa sau.
\end{itemize}
\end{tcolorbox}
