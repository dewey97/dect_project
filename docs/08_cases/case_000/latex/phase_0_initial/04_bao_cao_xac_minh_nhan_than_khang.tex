\section*{I. HOÀN CẢNH GIA ĐÌNH \& NHÂN THÂN}
\begin{itemize}
    \item \textbf{Bố mẹ:} Đã qua đời trong tai nạn giao thông nhiều năm trước.
    \item \textbf{Ông nội:} Cụ Nguyễn Văn Thành (chủ nhà gốc) vừa qua đời cách đây 3 tháng.
    \item \textbf{Họ hàng trực tiếp:} Trần Ngọc Mai (36 tuổi, em họ, con cô ruột).
    \item \textbf{Tình trạng hôn nhân:} Độc thân.
\end{itemize}

\section*{II. ĐẶC ĐIỂM TÍNH CÁCH \& MỐI QUAN HỆ}
\begin{enumerate}
    \item \textbf{Tính cách:} Nóng nảy, bốc đồng, từ nhỏ thích o ép, bắt nạt người khác.
    \item \textbf{Quan hệ tiền bạc:} Lối sống ăn tiêu bốc đồng. Đang làm hồ sơ nhận trọn vẹn tiền đền bù giải tỏa nhà đất (trên 2 tỷ đồng).
    \item \textbf{Quan hệ tình cảm:} Trăng hoa. Vừa chủ động phớt lờ đòi chia tay bạn gái cũ (Trần Thị Hà) để quen nhân tình mới.
    \item \textbf{Mâu thuẫn nổi bật:}
    \begin{itemize}
        \item Tranh chấp di chúc thừa kế nảy lửa với em họ Trần Ngọc Mai.
        \item Chèn ép, tống tiền Lê Quang Vũ bằng giấy nợ và bản vẽ đo đạc khống.
        \item Dùng bí mật tai nạn quá khứ đe dọa ép Tùng và Hà nhượng quyền thừa kế.
    \end{itemize}
\end{enumerate}
