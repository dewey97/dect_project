\section*{I. THÔNG TIN LÝ LỊCH VÀ HOÀN CẢNH GIA ĐÌNH}
\begin{itemize}[leftmargin=1.5cm]
    \item \textbf{Họ và tên:} NGUYỄN VĂN KHANG. Tên gọi khác: Khang "Đen".
    \item \textbf{Ngày sinh:} 15/06/1990. Nguyên quán: Nam Định.
    \item \textbf{Hoàn cảnh gia đình:} Bố mẹ mất sớm trong tai nạn giao thông năm 2005. Khang sống cùng ông nội (cụ Nguyễn Văn Thành) tại nhà số 14 Bờ Sông. Cụ Thành mới qua đời tháng 04/2026.
    \item \textbf{Hiện trạng tài sản:} Khang đang đứng tên quản lý căn nhà 75m2 thuộc diện sắp giải tỏa đền bù dự án Cảng sông (giá trị ước tính 3.5 tỷ đồng).
\end{itemize}

\section*{II. LỐI SỐNG VÀ CÁC MỐI QUAN HỆ PHỨC TẠP}
\begin{enumerate}[label=\arabic*., leftmargin=1.5cm]
    \item \textbf{Lối sống \& Tài chính:} Khang không có nghề nghiệp ổn định, ăn tiêu bốc đồng, thường xuyên cờ bạc bóng đá. Đang có nhiều khoản nợ nần ngoài xã hội.
    \item \textbf{Quan hệ tình cảm:} Khang sống trăng hoa. Đã yêu Trần Thị Hà được 3 năm nhưng gần đây phớt lờ Hà và lén lút quen tình nhân mới (Linh).
    \item \textbf{Mâu thuẫn đất đai & Tống tiền:} Khang mâu thuẫn gay gắt với em họ Trần Ngọc Mai về di chúc thừa kế của ông nội. Đồng thời nắm giữ bí mật quá khứ để tống tiền Tùng và Vũ.
\end{enumerate}
